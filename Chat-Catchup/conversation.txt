CODE
using UnityEngine;

public class SciFiRoomLight : MonoBehaviour
{
	[SerializeField] private Light mainLight;
	[SerializeField] private float rotationSpeed = 35f;

	private void Start()
	{
		mainLight.type = LightType.Directional;
		mainLight.intensity = 1.2f;
		mainLight.shadows = LightShadows.Soft;
	}

	private void Update()
	{
		float horizontal = Input.GetAxis("Horizontal");
		float vertical = Input.GetAxis("Vertical");
		transform.Rotate(Vector3.up, horizontal * rotationSpeed * Time.deltaTime);
		transform.Rotate(Vector3.right, -vertical * rotationSpeed * Time.deltaTime);
	}
}

SETUP
Create a Directional Light, place the focal object where it can receive light, and make sure the room has a floor and walls. Attach the script to an empty GameObject, assign the Directional Light, then press Play and use the arrow keys. Later, test the same scene on Meta Quest and tune the shadows for performance.

FIRST LESSON
1. Build one small room with simple walls, a floor, and one focal object.
2. Add one main light to create the sci-fi mood and clear shadows.
3. Bake the room's lighting where possible instead of relying on many real-time lights.
4. Test on the Meta Quest and check that the scene stays smooth before adding more detail.
